# ===================================================
# 交换庄园 — 支持七类住所
# ===================================================
zz_swap_domicile_interaction = {
    category = interaction_category_diplomacy
    interface_priority = 160
    common_interaction = yes
    icon = icon_vassal

    desc = zz_swap_domicile_interaction_desc

    ai_maybe = no
    auto_accept = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes

    # MOD 步骤1 (2026-07-20): 仅对"拥有庄园的封臣及下属"显示
    is_shown = {
        zz_md_vassal_swap_enabled = yes
        scope:actor = { has_domicile = yes }            # 发起者也必须拥有庄园才能交换
        scope:recipient = {
            this != scope:actor                         # 不能与自己交换
            is_vassal_or_below_of = scope:actor         # 封臣及下属
            has_domicile = yes                          # 对方也拥有庄园
        }
    }

    # Both owners must have the same supported domicile type.
    is_valid_showing_failures_only = {
        zz_md_vassal_swap_enabled = yes
        custom_description = { text = zz_swap_unsupported_type zz_swap_supported_types = yes }
        custom_description = { text = zz_swap_type_mismatch zz_domicile_type_matches = yes }
    }

    on_accept = {
        if = {
            limit = { zz_md_vassal_swap_enabled = yes }
            scope:actor = {
                if = { limit = { domicile ?= { is_domicile_type = estate } } trigger_event = zz_swap_estate.0001 }
                else_if = { limit = { domicile ?= { is_domicile_type = east_asian_estate } } trigger_event = zz_swap_domicile.0001 }
                else_if = { limit = { domicile ?= { is_domicile_type = japanese_manor } } trigger_event = zz_swap_japanese.0001 }
                else_if = { limit = { domicile ?= { is_domicile_type = camp } } trigger_event = zz_md_vassal.0001 }
                else_if = { limit = { domicile ?= { is_domicile_type = yurt } } trigger_event = zz_md_vassal.0002 }
                else_if = { limit = { domicile ?= { is_domicile_type = ecclesiastical_domicile } } trigger_event = zz_md_vassal.0003 }
                else_if = { limit = { domicile ?= { is_domicile_type = merchant_company } } trigger_event = zz_md_vassal.0004 }
            }
        }
    }

    ai_accept = { base = 100 }
    ai_will_do = { base = 0 }
}
