# R25 current1.20.0.4 native specialized transaction callbacks retained.
zz_md_grant_domicile_title_interaction = {

	category = interaction_category_vassal
	common_interaction = yes

	desc = zz_md_grant_domicile_title_interaction_desc
	icon = grant_titles_interaction

	special_interaction = grant_titles_interaction
	interface = grant_titles
	target_type = title
	target_source = puppet_or_actor
	target_filter = domain_titles
	interface_priority = 60

	# actor character giving the titles
	# recipient character receiving the titles

	is_shown = {
		scope:puppet_or_actor != scope:recipient
		scope:puppet_or_actor ?= {
			NOT = {
				government_has_flag = cannot_be_vassal_or_liege
			}
		}
		scope:recipient = {
			OR = {
				target_is_liege_or_above = scope:puppet_or_actor
				is_pool_guest_of = scope:puppet_or_actor
				AND = {
					scope:puppet_or_actor.faith.religious_head ?= scope:recipient
					top_liege ?= this
				}
			}
		}
		
		trigger_if = {
			limit = {
				scope:puppet_or_actor = { government_has_flag = government_is_nomadic }
			}
			scope:recipient = { NOT = { government_has_flag = government_is_herder } }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = can_be_granted_titles_interaction_tt
				OR = {
					can_be_granted_titles_by = { RULER = scope:puppet_or_actor }
					can_be_granted_theocratic_titles_by = { RULER = scope:puppet_or_actor }
					AND = {
						is_diarch_of_target = scope:puppet_or_actor
						scope:puppet_or_actor = { has_diarchy_active_parameter = diarchy_is_co_rulership }
					}
				}
			}
			NOT = { is_at_war_with = scope:puppet_or_actor }
			OR = { # Monks are disinherited from secular land; the exemption keys on the granter's government because the titles are picked after this check
				NOT = { has_trait = devoted }
				scope:puppet_or_actor = { government_has_flag = government_is_theocracy }
			}
			bp2_valid_for_standard_interactions_trigger = yes
			trigger_if = {
				limit = { is_ruler = no }
				is_imprisoned = no
			}
			# DON'T TAKE OTHER PLAYERS' ACCLAIMED KNIGHTS
			custom_description = {
				text = grant_title_stealing_acclaimed_knight
				subject = scope:recipient
				NOR = {
					AND = {
						is_acclaimed = yes
						liege ?= {
							is_ai = no
							this != scope:puppet_or_actor
						}
					}
					AND = {
						is_acclaimed = yes
						scope:actor = { is_ai = yes }
					}
				}
			}
			# Gallivanters won't accept additional responsibilities.
			NOT = { has_trait = gallivanter }
			# TGP JAPAN
			tgp_japan_grant_titles_restriction_trigger = {
				GRANTER = scope:puppet_or_actor
				GRANTEE = scope:recipient
			}
		}
		custom_description = {
			text = is_not_theocratic_court_chaplain
			subject = scope:recipient
			NAND = {
				scope:puppet_or_actor.rite = {
					rite_has_doctrine = doctrine_theocracy_temporal
				}
				scope:puppet_or_actor = { NOT = { government_has_flag = government_is_theocracy } }
				scope:recipient = {
					rite = {
						rite_has_doctrine = doctrine_theocracy_temporal
					}
					has_council_position = councillor_court_chaplain
				}
				scope:puppet_or_actor.faith.religious_head != scope:recipient
			}
		}
		scope:puppet_or_actor = {
			any_held_title = {
				count > 1
			}
		}
	}

	can_send = {
		trigger_if = {
			limit = {
				scope:puppet_or_actor = {
					highest_held_title_tier >= tier_kingdom
					has_diarchy_active_parameter = primeminister_requires_duchy
					diarch = scope:recipient
				}
			}
			custom_tooltip = {
				text = primeminister_requires_duchy.tt.blocked_grant
				any_in_list = {
					list = target_titles
					tier >= tier_duchy
				}
			}
		}
	
		any_in_list = { list = target_titles always = yes }
        NOT = { any_in_list = { list = target_titles NOT = { zz_md_transfer_title_eligible = yes holder = scope:puppet_or_actor } } }
    }

	can_be_picked_title = { scope:target = { zz_md_transfer_title_eligible = yes holder = scope:puppet_or_actor } }

	auto_accept = yes

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0110
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:puppet_or_actor = {
					government_has_flag = government_is_nomadic
					vassal_limit_available <= 0
					is_independent_ruler = yes
					scope:recipient = {
						NOT = { is_vassal_of = scope:puppet_or_actor }
					}
				}
			}
			custom_tooltip = grant_title_nomad_and_at_vassal_limit
		}
		scope:recipient = {
			add_character_flag = {
				flag = domicile_culture_and_faith_check
				days = 1
			}
		}
		if = {
			limit = {
				scope:recipient = {
					is_acclaimed = yes
					liege = {
						this = scope:puppet_or_actor
					}
				}
				scope:target = {
					tier > tier_barony
				}
			}
			custom_tooltip = grant_title_landing_acclaimed_knight_warning
		}

		grant_title_holy_order_patronage_warning_effect = yes
		scope:recipient = {
			if = {
				limit = {
					top_liege = this
					save_temporary_scope_as = recipient_is_independent
				}
			}
			if = {
				limit = {
					NOT = {
						has_relation_friend = scope:puppet_or_actor
					}
				}
				if = {
					# then verify the right struggle phase
					limit = {
						is_diff_faith_or_culture_trigger = {
							CHAR = scope:puppet_or_actor
							STATUS = involved
						}
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = struggle_grant_titles_diff_faith_culture_to_leads_to_friendship
						}
					}
					progress_towards_friend_effect = {
						REASON = friend_granted_title
						CHARACTER = scope:puppet_or_actor
						OPINION = 0
					}
				}
			}
		}

		if = {
			limit = {
				OR = {
					any_in_list = {
						list = target_titles
						tier = tier_county
						culture = scope:recipient.culture
						NOR = {
							culture = scope:puppet_or_actor.culture
							recent_history = {
								type = granted
								years = 10
							}
						}
					}
					any_in_list = {
						list = target_titles
						tier >= tier_duchy
						title_capital_county = {
							culture != scope:puppet_or_actor.culture
							culture = scope:recipient.culture
						}
						NOT = {
							recent_history = {
								type = granted
								years = 10
							}
						}
					}
				}
			}
			scope:recipient.culture = {
				change_cultural_acceptance = {
					target = scope:puppet_or_actor.culture
					value = grant_title_cultural_acceptance_impact
					desc = cultural_acceptance_gain_granted_title
				}
			}
		}
		if = {
			limit = {
				any_in_list = {
					list = target_titles
					tier = tier_county
				}
				scope:puppet_or_actor.culture = { has_cultural_parameter = grants_to_lowborns_bonuses }
				scope:recipient = {
					is_lowborn = yes
				}
			}
			every_in_list = {
				list = target_titles
				limit = { tier = tier_county }
				change_county_control = 30
			}
			scope:recipient = {
				add_trait = peasant_leader
			}
		}
		if = {
			limit = {
				any_in_list = {
					list = target_titles
					tier >= tier_county
				}
				scope:puppet_or_actor.culture = { has_cultural_parameter = landing_house_members_give_prestige }
				exists = scope:puppet_or_actor.house
				exists = scope:recipient.house
				scope:puppet_or_actor.house = scope:recipient.house
			}
			scope:puppet_or_actor = {
				add_prestige = {
					value = 0
					every_in_list = {
						list = target_titles
						limit = { tier = tier_hegemony }
						add = monumental_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_empire }
						add = massive_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_kingdom }
						add = major_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_duchy }
						add = medium_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_county }
						add = minor_prestige_gain
					}
				}
			}
		}
		if = {
			limit = {
				any_in_list = {
					list = target_titles
					tier >= tier_county
				}
				scope:puppet_or_actor.culture = { has_cultural_parameter = landing_house_members_gives_renown }
				exists = scope:puppet_or_actor.house
				exists = scope:recipient.house
				scope:puppet_or_actor.house = scope:recipient.house
			}
			scope:puppet_or_actor.dynasty = {
				add_dynasty_prestige = {
					value = 0
					every_in_list = {
						list = target_titles
						limit = { tier = tier_hegemony }
						add = scope:puppet_or_actor.dynasty.monumental_dynasty_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_empire }
						add = scope:puppet_or_actor.dynasty.massive_dynasty_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_kingdom }
						add = scope:puppet_or_actor.dynasty.major_dynasty_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_duchy }
						add = scope:puppet_or_actor.dynasty.medium_dynasty_prestige_gain
					}
					every_in_list = {
						list = target_titles
						limit = { tier = tier_county }
						add = scope:puppet_or_actor.dynasty.minor_dynasty_prestige_gain
					}
				}
			}
		}

		# Iterate over all given titles
		#  - Add opinion modifiers
		#  - Collect titles that discontent reduction can later be derived from
		every_in_list = {
			list = target_titles
			save_temporary_scope_as = this_title
			if = {
				limit = {
					exists = scope:landed_title
				}
				scope:recipient = {
					if = {
						limit = {
							scope:this_title.tier = tier_barony
						}

						hidden_effect = {
							add_opinion = {
								target = scope:puppet_or_actor
								modifier = received_title_barony
							}
						}

						scope:this_title = { add_to_temporary_list = titles_to_grant }
					}

					else_if = {
						limit = {
							scope:this_title.tier = tier_county
						}

						hidden_effect = {
							add_opinion = {
								target = scope:puppet_or_actor
								modifier = received_title_county
							}
							if = {
								limit = {
									is_lowborn = yes
								}
								scope:puppet_or_actor = {
									if = {
										limit = {
											scope:actor = { is_ai = no }
										}
										every_vassal = {
											vassal_stance = courtly
											add_opinion = {
												target = scope:puppet_or_actor
												modifier = courtly_lowborn_grant_opinion
												opinion = -5
											}
										}
									}
									else = { # Can't really stop the AI from landing lowborns, so throttle it a bit so that great conquerors don't annihilate their opinion
										random = {
											chance = 25
											every_vassal = {
												vassal_stance = courtly
												add_opinion = {
													target = scope:puppet_or_actor
													modifier = courtly_lowborn_grant_opinion
													opinion = -5
												}
											}
										}
									}
								}
							}
						}

						scope:this_title = { add_to_temporary_list = titles_to_grant }
					}

					else_if = {
						limit = {
							scope:this_title.tier = tier_duchy
						}

						hidden_effect = {
							add_opinion = {
								target = scope:puppet_or_actor
								modifier = received_title_duchy
							}
							if = {
								limit = {
									is_lowborn = yes
								}
								scope:puppet_or_actor = {
									every_vassal = {
										vassal_stance = courtly
										add_opinion = {
											target = scope:puppet_or_actor
											modifier = courtly_lowborn_grant_opinion
											opinion = -20
										}
									}
								}
							}
						}

						scope:this_title = { add_to_temporary_list = titles_to_grant }
					}

					else_if = {
						limit = {
							scope:this_title.tier = tier_kingdom
						}

						hidden_effect = {
							add_opinion = {
								target = scope:puppet_or_actor
								modifier = received_title_kingdom
							}
							if = {
								limit = {
									is_lowborn = yes
								}
								scope:puppet_or_actor = {
									every_vassal = {
										vassal_stance = courtly
										add_opinion = {
											target = scope:puppet_or_actor
											modifier = courtly_lowborn_grant_opinion
											opinion = -50
										}
									}
								}
							}
						}

						scope:this_title = { add_to_temporary_list = titles_to_grant }
					}

					else_if = {
						limit = {
							scope:this_title.tier >= tier_empire
						}

						hidden_effect = {
							add_opinion = {
								target = scope:puppet_or_actor
								modifier = received_title_empire
							}
							if = {
								limit = {
									is_lowborn = yes
								}
								scope:puppet_or_actor = {
									every_vassal = {
										vassal_stance = courtly
										add_opinion = {
											target = scope:puppet_or_actor
											modifier = courtly_lowborn_grant_opinion
											opinion = -80
										}
									}
								}
							}
							if = {
								limit = {
									is_a_faction_member = yes
								}
								add_joined_faction_discontent = -60
							}
						}

						scope:this_title = { add_to_temporary_list = titles_to_grant }
					}
				}

				#Promised a vassal this title they had a claim on (vassal.2901)
				if = {
					limit = {
						scope:puppet_or_actor = {
							exists = var:was_promised_title
							var:was_promised_title = scope:recipient
						}
					}
					if = {
						limit = {
							scope:recipient = {
								exists = var:promised_title
							}
							scope:landed_title = scope:recipient.var:promised_title
						}
						scope:puppet_or_actor = {
							trigger_event = {
								id = vassal.2910
								days = { 7 14 }
							}
						}
					}
				}
			}
			if = {
				limit = {
					scope:recipient = {
						has_trait = bastard
					}
				}
				custom_tooltip = grant_title_to_bastard_dynasty_warning
			}
		} # every_in_list


		# The accumulated discontent reduction of the recipient
		scope:recipient = {
			if = {
				limit = {
					is_a_faction_member = yes
				}
				add_joined_faction_discontent = grant_title_opinion_sum_discontent_calc
			}
		}

		# Viziers may be booted from their job.
		scope:puppet_or_actor = {
			if = {
				limit = {
					# Assuming this is relevant at all.
					diarch ?= scope:recipient
					# And that scope:puppet_or_actor can do without a supervisor atm.
					can_leave_diarchy_trigger = yes
					# Aaaaand that their diarchy supports this type of thing.
					OR = {
						has_diarchy_parameter = primeminister_requires_county
						has_diarchy_parameter = primeminister_requires_duchy
					}
				}
				# Is this a duchy?
				if = {
					limit = {
						any_in_list = {
							list = target_titles
							tier >= tier_duchy
						}
					}
					custom_tooltip = grant_titles_interaction.tt.primeminister_title_grant.duchy_sufficient
					end_diarchy = yes
				}
				# Is this a county?
				else_if = {
					limit = {
						any_in_list = {
							list = target_titles
							tier = tier_county
						}
					}
					# Is that insufficient?
					if = {
						limit = { has_diarchy_active_parameter = primeminister_requires_duchy }
						custom_tooltip = grant_titles_interaction.tt.primeminister_title_grant.county_insufficient
					}
					# Is that sufficient?
					else = {
						custom_tooltip = grant_titles_interaction.tt.primeminister_title_grant.county_sufficient
						end_diarchy = yes
					}
				}
				# Is this a barony?
				else_if = {
					limit = {
						any_in_list = {
							list = target_titles
							count = all
							tier = tier_barony
						}
					}
					custom_tooltip = grant_titles_interaction.tt.primeminister_title_grant.barony_doesnt_end
				}
			}
		}

		if = {
			limit = {
				any_in_list = {
					list = target_titles
					tier >= tier_empire
				}
			}
			custom_tooltip = grant_title_modifier_stack_empires
		}
		if = {
			limit = {
				any_in_list = {
					list = target_titles
					tier = tier_kingdom
				}
			}
			custom_tooltip = grant_title_modifier_stack_kingdoms
		}
		if = {
			limit = {
				any_in_list = {
					list = titles_to_grant
					tier = tier_duchy
				}
			}
			custom_tooltip = grant_title_modifier_stack_duchies
		}
		if = {
			limit = {
				any_in_list = {
					list = titles_to_grant
					tier = tier_county
				}
			}
			custom_tooltip = grant_title_modifier_stack_counties
		}
		if = {
			limit = {
				any_in_list = {
					list = titles_to_grant
					tier = tier_barony
				}
			}
			custom_tooltip = grant_title_modifier_stack_baronies
		}
		if = {
			limit = {
				trigger_if = {
					limit = { scope:actor = { is_ai = no } }
					scope:puppet_or_actor = {
						any_vassal = {
							vassal_stance = courtly
						}
					}
				}
				scope:recipient = {
					is_lowborn = yes
				}
				any_in_list = {
					list = titles_to_grant
					tier >= tier_county
				}
			}
			custom_tooltip = grant_title_modifier_courtly_lowborn_grant_penalty
		}
		if = {
			limit = {
				scope:recipient = { is_a_faction_member = yes }
			}
			custom_tooltip = grant_title_modifier_stack_discontent
		}

		# Jealousy check
		if = {
			limit = {
				any_in_list = {
					list = titles_to_grant
					any_claimant = { grant_title_rivalry_trigger = yes }
				}
			}
			hidden_effect = {
				random = {
					chance = 25
					ordered_in_list = {
						list = titles_to_grant
						limit = {
							any_claimant = { grant_title_rivalry_trigger = yes }
						}
						order_by = tier
						save_scope_as = claim_title
						ordered_claimant = {
							limit = { grant_title_rivalry_trigger = yes }
							order_by = primary_title.tier
							save_scope_as = claim_vassal
							if = {
								limit = {
									scope:recipient = { is_lowborn = yes }
								}
								scope:recipient = { save_scope_as = lowborn_scope }
							}
							else_if = {
								limit = {
									scope:recipient = { has_claim_on = scope:claim_title }
								}
								scope:recipient = { save_scope_as = claimant_scope }
							}
							scope:puppet_or_actor = {
								trigger_event = {
									id = bp1_yearly.8070
									days = 5
								}
							}
						}
					}
				}
			}
		}

		#Stress impact
		scope:puppet_or_actor = {
			if = {
				limit = {
					any_in_list = {
						list = target_titles
						trigger_if = {
							limit = { tier = tier_barony }
							title_province.barony = {
								has_wrong_holding_type = no
							}
						}
						count >= 1
					}
				}

				# Generous characters enjoy giving away titles
				if = {
					limit = {
						has_trait = generous
					}
					add_stress = generous_title_grant_value
				}

				# Gnostics lose stress for giving away titles.
				if = {
					limit = {
						scope:puppet_or_actor.rite = {
							rite_has_parameter = granting_titles_gives_stress
						}
					}
					scope:puppet_or_actor = {
						add_stress = {
							value = 0

							# Base stress value scaled on title tier.
							every_in_list = {
								list = target_titles

								add = {
									value = 0

									if = {
										limit = { this.tier = tier_county }
										add = minor_stress_loss # 10 base
									}
									else_if = {
										limit = { this.tier = tier_duchy }
										add = {
											value = medium_stress_loss
											multiply = 1.25 # 25 base
										}
									}
									else_if = {
										limit = { this.tier = tier_kingdom }
										add = {
											value = major_stress_loss
											multiply = 2 # 80 base
										}
									}
									else_if = {
										limit = { this.tier >= tier_empire }
										add = {
											value = monumental_stress_loss
											multiply = 2 # 200 base
										}
									}

									# Lose more stress for giving away titles of your primary tier; lose less stress for giving away titles far below your primary in tier.
									multiply = {
										value = this.tier
										add = 1
										divide = scope:puppet_or_actor.primary_title.tier
									}

									# Round to the nearest multiple of 5.
									divide = 5
									round = yes
									multiply = 5

									# For a Duke, this will be x1.25 for Duchies (-30) and x1 for Counties(-10)
									# For a King, this will be x1.25 for a Kingdom (-100), x1 for a Duchy (-25) and x0.75 for Counties(-10)
									# For an Emperor, this will be x1.25 for an Empire(-200), x1 for a Kingdom(-80), x0.75 for a Duchy(-20) and x0.4 for a Counties (-5)
								}
							}
						}
					}
				}

				# Greedy & ambitious ruler gain stress for giving away titles if they would normally be able to keep them.
				if = {
					limit = {
						# Pre-filter

							has_trait = greedy

									}

							domain_size = domain_limit

				custom_tooltip = runestone_grant_title_warning

						has_county_modifier = county_raised_runestone_modifier
					}
					NOT = { scope:recipient.dynasty = var:ancestor_to_bury.dynasty } #We only have to change something if the new holder is of a different dynasty
				}
				# Actually removed in religious_decision.0312
				custom_tooltip = runestone_grant_title_warning
			}

						has_dynasty_perk = ep1_culture_legacy_3

					scope:recipient = {
						is_playable_character = no
						culture != scope:puppet_or_actor.culture
						any_in_list = {
							list = target_titles
							OR = {
								tier = tier_county
								tier = tier_barony
							}
							culture = scope:recipient.culture

					if = {
						limit = {
							can_add_hook = {
								target = scope:recipient
								type = favor_hook
							}
						}

							type = favor_hook
							target = scope:recipient

						right_icon = scope:recipient

						type = event_toast_effect_neutral
						title = grant_titles_interaction_notification

						left_icon = scope:puppet_or_actor
						right_icon = scope:recipient

						if = {
							limit = {
								exists = dynasty

									STATUS = involved

						right_icon = scope:recipient

				else_if = {

					days = 365

		#Opinion loss with underlings who think they should get the title you're granting
		if = {
			limit = {
				#AI will usually grant title to de jure holders, it's players who could use this steering
				scope:actor = {
					is_ai = no
				}
			}
			if = {
				limit = {
					any_in_list = {
						list = titles_to_grant
						any_claimant = {
							this = scope:recipient
						}
					}
				}
				ordered_in_list = {
					order_by = tier
					list = titles_to_grant
					limit = {
						any_claimant = {
							this = scope:recipient
						}
					}
					save_scope_as = highest_claim
				}
				scope:recipient = { claimant_jealousy_immunity_effect = yes }
			}
			every_in_list = {
				list = titles_to_grant
				save_scope_as = title_granted
				scope:puppet_or_actor = {
					title_granting_controversy_effect = {
						TITLE = scope:title_granted
						INTERACTION = grant_titles_interaction
					}
				}
			}
			every_in_list = {
				list = strong_claimants
				limit = {
					NOT = { this = scope:recipient }
				}
				save_scope_as = claimer
				ordered_in_list = {
					order_by = tier
					list = titles_to_grant
					limit = {
						any_claimant = {
							this = scope:claimer
						}
					}
					save_scope_as = highest_denied_claim
				}
				claimant_rivalry_mercy_effect = yes
			}
		}
	}

			scope:puppet_or_actor.domain_size < scope:puppet_or_actor.domain_limit
			scope:puppet_or_actor.primary_title.tier > tier_county
			trigger_if = {
				limit = {
					scope:puppet_or_actor = {
						title_revocation_is_tyrannical_trigger = { VASSAL = scope:recipient }
						domain_size > 3
					}
				}
				scope:puppet_or_actor = {
					NOR = {
						has_trait = generous
						has_trait = just
					}
				}
			}
			exists = scope:puppet_or_actor.primary_title.title_capital_county
			OR = {
				scope:puppet_or_actor.primary_title.title_capital_county.holder = scope:puppet_or_actor # You have your de jure capital already
				NOT = {
					scope:puppet_or_actor.primary_title.title_capital_county.holder.top_liege = scope:puppet_or_actor.top_liege # Or your De Jure capital isn't in your realm
				}
			}
			scope:landed_title.tier = tier_county # Only Counties
			OR = { # The AI gives away wrong-holding counties almost instantly, so we restrict to counties with holdings they can hold
				scope:landed_title.title_province = { has_holding_type = castle_holding } # Everyone is valid for castles
				AND = {
					scope:puppet_or_actor = {
						government_has_flag = government_is_tribal
					}
					scope:landed_title.title_province = { has_holding_type = tribal_holding }
				}
				AND = {
					scope:puppet_or_actor = {
						government_has_flag = government_is_republic
					}
					scope:landed_title.title_province = { has_holding_type = city_holding }
				}
				AND = {
					scope:puppet_or_actor = {
						OR = {
							government_has_flag = government_is_clan
							government_has_flag = government_is_theocracy
						}
					}
					scope:landed_title.title_province = { has_holding_type = church_holding }
				}
			}
			OR = { # And the capital duchy, if it exists at all
				AND = {
					exists = scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege.holder
					scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege.holder = scope:puppet_or_actor
				}
				AND = {
					exists = scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege.holder
					scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege.holder.top_liege != scope:puppet_or_actor.top_liege
				}
				scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege = {
					is_title_created = no
				}
			}

			scope:landed_title.de_jure_liege = scope:puppet_or_actor.primary_title.title_capital_county.de_jure_liege # Under the same Duchy as your capital

			scope:recipient = { # Not from players
				is_ai = yes
			}

			OR = { # Tyranny tolerance is based on rationality
				AND = {
					scope:puppet_or_actor.ai_rationality < 0
					scope:puppet_or_actor.tyranny <= 15 # Don't go crazy with tyranny, just a bit is fine
				}
				AND = {
					scope:puppet_or_actor.ai_rationality >= 0
					scope:puppet_or_actor.ai_rationality < 50
					scope:puppet_or_actor.tyranny <= 5 # Don't go crazy with tyranny, just a bit is fine
				}
				AND = {
					scope:puppet_or_actor.ai_rationality >= 50
					scope:puppet_or_actor.tyranny <= 0 # Don't go crazy with tyranny, just a bit is fine
				}
				AND = {
					scope:puppet_or_actor.ai_rationality >= 0
					scope:recipient = {
						is_imprisoned_by = scope:puppet_or_actor
					}
					scope:puppet_or_actor = { has_revoke_title_reason = scope:recipient }
				}
				AND = {
					scope:puppet_or_actor.ai_rationality < 0
					scope:recipient = {
						is_imprisoned_by = scope:puppet_or_actor
					}
				}
			}
			# To do this, the target must be weak or willing in some way
			OR = {
				scope:puppet_or_actor.current_military_strength > scope:recipient.current_military_strength # Must be stronger
				scope:recipient = {
					is_imprisoned_by = scope:puppet_or_actor
				}
			}
			scope:recipient = {
				OR = {
					AND = {
						scope:recipient.primary_title.tier <= tier_county
						scope:puppet_or_actor.primary_title.tier >= tier_duchy
					}
					has_trait = craven
					has_trait = content
					has_trait = trusting
					has_dread_level_towards = {
						target = scope:puppet_or_actor
						level >= 1
					}
					opinion = { target = scope:puppet_or_actor value >= 40 }
					is_imprisoned_by = scope:puppet_or_actor
					scope:puppet_or_actor = { has_revoke_title_reason = scope:recipient }
				}
			}

			add = 1100
		}
		# Revoke counties we're dejure liege of if below domain limit
		modifier = {
			scope:puppet_or_actor.domain_size < scope:puppet_or_actor.domain_limit
			scope:landed_title.tier = tier_county
			scope:landed_title.de_jure_liege.holder = scope:puppet_or_actor
			add = 500
		}
		# Revoke appropriate baronies if below domain limit, it's tyranny free
		modifier = {
			scope:puppet_or_actor.domain_size < scope:puppet_or_actor.domain_limit
			scope:recipient = { highest_held_title_tier = tier_barony }
			scope:landed_title.tier = tier_barony
			OR = {
				scope:landed_title.title_province = { has_holding_type = castle_holding }
				scope:landed_title.title_province = { has_holding_type = temple_citadel_holding }
				AND = {
					scope:landed_title.title_province = { has_holding_type = church_holding }
					OR = {
						scope:puppet_or_actor = { government_has_flag = government_is_clan }
						scope:puppet_or_actor = { government_has_flag = government_is_theocracy }
						scope:puppet_or_actor = { government_has_flag = government_is_mandala }
						scope:puppet_or_actor.rite = { rite_has_parameter = lay_clergy }
					}
				}
				AND = {
					scope:landed_title.title_province = { has_holding_type = city_holding }
					scope:puppet_or_actor = { government_has_mechanic = administrative }
				}
			}
			add = 1100
		}
		# Revoke from vassals of the wrong faith if you can do it without tyranny
		modifier = {
			scope:puppet_or_actor = {
				NOT = {
					title_revocation_is_tyrannical_trigger = { VASSAL = scope:recipient }
				}
			}
			scope:puppet_or_actor.faith != scope:recipient.faith
			scope:recipient.primary_title.tier > tier_barony
			OR = {
				scope:puppet_or_actor.faith.religion != scope:recipient.faith.religion
				scope:puppet_or_actor.ai_zeal >= low_negative_ai_value
				NOT = {
					scope:puppet_or_actor.rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
				}
			}
			OR = { # Chill a bit until you've dealth with the ones already in your prison
				scope:recipient = {
					is_imprisoned_by = scope:puppet_or_actor
				}
				NOT = {
					any_vassal = {
						faith != scope:puppet_or_actor.faith
						primary_title.tier > tier_barony
						is_imprisoned_by = scope:puppet_or_actor
						time_in_prison = { years < 1 }
					}
				}
			}
			add = 100
		}
		# Revoke from characters you dislike if you can do it without tyranny
		modifier = {
			scope:puppet_or_actor = {
				NOT = {
					title_revocation_is_tyrannical_trigger = { VASSAL = scope:recipient }
				}
				OR = {
					has_relation_rival = scope:recipient
					has_relation_nemesis = scope:recipient
				}
			}
			add = 100
		}
		# Slight preference for higher-tier titles
		modifier = {
			add = scope:landed_title.tier
		}
		# Do not revoke titles from certain characters
		modifier = {
			scope:puppet_or_actor = {
				OR = {
					has_relation_friend = scope:recipient
					has_relation_best_friend = scope:recipient
					has_relation_lover = scope:recipient
					has_relation_soulmate = scope:recipient
					is_parent_of = scope:recipient
					is_grandparent_of = scope:recipient
					is_great_grandparent_of = scope:recipient
				}
			}
			add = -5000
		}
		# Do not revoke titles when the chance of success if exceedingly low
		modifier = {
			NOR = {
				scope:recipient = {
					is_imprisoned_by = scope:puppet_or_actor
				}
				scope:recipient = {
					highest_held_title_tier = tier_barony
				}
			}
			scope:puppet_or_actor = {
				OR = {
					AND = {
						has_treasury = no
						OR = {
							gold < minimum_ai_gold_value_for_tyranny_wars
							AND = {
								scope:puppet_or_actor.current_military_strength < scope:recipient.current_military_strength
								gold < comfortable_ai_gold_value_for_tyranny_wars
							}
						}
					}
					AND = {
						has_treasury = yes
						OR = {
							gold < minimum_ai_treasury_value_for_tyranny_wars
							AND = {
								scope:puppet_or_actor.current_military_strength < scope:recipient.current_military_strength
								treasury < comfortable_ai_treasury_value_for_tyranny_wars
							}
						}
					}
				}
			}
			add = -2000
		}

		modifier = { # The AI will always use a Hook if they can
			scope:hook = yes
			add = 1
		}

		modifier = { # The AI shouldn't ruin the player's fun
			add = -5000
			scope:recipient = { has_character_flag = is_party_baron }
		}

		# Revoke from prisoners whose deaths would diminish the realm through inheritance
		modifier = {
			add = 1000
			scope:recipient = {
				is_vassal_or_below_of = scope:puppet_or_actor
				any_held_title = {
					is_landless_type_title = no
					tier >= tier_county
					current_heir ?= {
						is_ruler = yes
						is_landed = yes
						top_liege != scope:puppet_or_actor.top_liege
					}
				}
			}
		}

		# Clan Government: Avoid revoking titles from house members if the AI wants to have high unity
		modifier = {
			scope:puppet_or_actor = {
				government_has_flag = government_is_clan
				exists = house
				ai_wants_high_unity = yes
			}
			scope:recipient = {
				government_has_flag = government_is_clan
				house ?= scope:puppet_or_actor.house
			}
			add = -1000
		}		# Admin Government: Do not revoke titles from your house unless it's really needed
		modifier = {
			scope:puppet_or_actor = { government_has_mechanic = administrative }
			exists = scope:recipient.house
			scope:puppet_or_actor.house ?= scope:recipient.house
			add = -500
		}
		# Japanese Admin wants to revoke Japanese Feudal if they have a reason
		modifier = {
			add = 5000
			scope:puppet_or_actor = {
				government_has_flag = government_is_japan_administrative
				exists = house
				has_revoke_title_reason = scope:recipient
				NOT = { has_relation_friend = scope:recipient }
			}
			scope:recipient = {
				government_has_flag = government_is_japan_feudal
				is_imprisoned = yes
				house != scope:puppet_or_actor.house
			}
		}
		# And Japanese Feudal wants to revoke Japanese Admin if they have a reason
		modifier = {
			add = 5000
			scope:puppet_or_actor = {
				government_has_flag = government_is_japan_feudal
				exists = house
				has_revoke_title_reason = scope:recipient
				NOT = { has_relation_friend = scope:recipient }
			}
			scope:recipient = {
				government_has_flag = government_is_japan_administrative
				is_imprisoned = yes
				house != scope:puppet_or_actor.house
			}
		}
		# TGP Be more careful with blocs generally
		house_bloc_tyranny_war_modifiers = yes

		modifier = { # Avoid triggering Tyranny wars mid-war
			factor = 0
			scope:puppet_or_actor = {
				is_at_war = yes
			}
			NOR = {
				scope:recipient = {
					is_imprisoned_by = scope:puppet_or_actor
				}
				scope:recipient = {
					highest_held_title_tier = tier_barony
				}
			}
		}

		# Don't take a title off anyone when refusing would raise half the realm against us. A prisoner, baron and lawful revocations are exempt, because they don't cause civil wars
		modifier = {
			factor = 0
			scope:recipient = {
				NOR = {
					is_imprisoned_by = scope:puppet_or_actor
					highest_held_title_tier = tier_barony
					government_has_flag = government_is_ecclesiastical
				}
			}
			scope:puppet_or_actor = {
				title_revocation_is_tyrannical_trigger = { VASSAL = scope:recipient }
				revoke_refusal_war_would_be_massive_trigger = { VASSAL = scope:recipient }
			}
		}

		# Recipient Ecclesiastical: Don't revoke if not excommunicated
		modifier = {
			factor = 0
			scope:recipient = {
				government_has_flag = government_is_ecclesiastical
				NOT = { has_trait = excommunicated }
			}
			scope:puppet_or_actor = {
				title_revocation_is_tyrannical_trigger = { VASSAL = scope:recipient }
				revoke_refusal_war_would_be_massive_trigger = { VASSAL = scope:recipient }
			}
		}

		modifier = { # Avoid triggering Tyranny wars mid-grand activity
			factor = 0
			scope:puppet_or_actor = {
				has_ongoing_grand_activity_trigger = yes
			}
			NOR = {
				scope:recipient = {
					is_imprisoned_by = scope:puppet_or_actor
				}
				scope:recipient = {
					highest_held_title_tier = tier_barony
				}
			}
		}

		# Admin rulers should be less likely to revoke titles without a revocation reason, as they should prefer to depose governors
		modifier = {
			scope:puppet_or_actor = {
				government_has_mechanic = administrative
				NOT = { has_revoke_title_reason = scope:recipient }
			}
			scope:recipient = {
				NOT = { highest_held_title_tier = tier_barony }
				government_has_mechanic = administrative
				trigger_if = {
					limit = {
						scope:puppet_or_actor = { has_government = japan_administrative_government }
					}
					highest_held_title_tier >= tier_county
				}
				trigger_else = { highest_held_title_tier >= tier_duchy }
			}
			factor = { # Honorable AI's will never revoke titles without a reason
				value = 0
				if = {
					limit = {
						ai_honor < 0
					}
					add = 0.5
				}
			}
		}

		#Don't revoke titles from the frankokratia leader
		modifier = {
			scope:recipient = {
				any_owned_story = {
					type = frankokratia_story
					NOT = { has_variable = franko_war_complete }
				}
			}
			factor = 0
		}
