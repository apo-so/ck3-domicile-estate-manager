# 调用方: common/character_interactions/zz_swap_domicile_interaction.txt
# 可用 scope: scope:actor (发起者/玩家), scope:recipient (目标封臣)

# 同类型判定: actor 与 recipient 的庄园类型一致
zz_domicile_type_matches = {
    OR = {
        AND = { scope:actor = { domicile ?= { is_domicile_type = estate } } scope:recipient = { domicile ?= { is_domicile_type = estate } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = east_asian_estate } } scope:recipient = { domicile ?= { is_domicile_type = east_asian_estate } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = japanese_manor } } scope:recipient = { domicile ?= { is_domicile_type = japanese_manor } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = camp } } scope:recipient = { domicile ?= { is_domicile_type = camp } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = yurt } } scope:recipient = { domicile ?= { is_domicile_type = yurt } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = ecclesiastical_domicile } } scope:recipient = { domicile ?= { is_domicile_type = ecclesiastical_domicile } } }
        AND = { scope:actor = { domicile ?= { is_domicile_type = merchant_company } } scope:recipient = { domicile ?= { is_domicile_type = merchant_company } } }
    }
}

zz_swap_can_execute = {
    zz_md_vassal_swap_enabled = yes
    exists = scope:actor
    exists = scope:recipient
    exists = scope:zz_md_prepared_a
    exists = scope:zz_md_prepared_b
    NOT = { scope:zz_md_prepared_a = scope:zz_md_prepared_b }
    scope:actor = { is_alive = yes domicile = scope:zz_md_prepared_a }
    scope:recipient = { is_alive = yes this != scope:actor is_vassal_or_below_of = scope:actor domicile = scope:zz_md_prepared_b }
    scope:zz_md_prepared_a = { owner = scope:actor }
    scope:zz_md_prepared_b = { owner = scope:recipient }
    zz_domicile_type_matches = yes
}

zz_swap_supported_types = {
    scope:actor = { domicile = { OR = { is_domicile_type = estate is_domicile_type = east_asian_estate is_domicile_type = japanese_manor is_domicile_type = camp is_domicile_type = yurt is_domicile_type = ecclesiastical_domicile is_domicile_type = merchant_company } } }
    scope:recipient = { domicile = { OR = { is_domicile_type = estate is_domicile_type = east_asian_estate is_domicile_type = japanese_manor is_domicile_type = camp is_domicile_type = yurt is_domicile_type = ecclesiastical_domicile is_domicile_type = merchant_company } } }
}

# Shared by the vassal interaction and its event-time revalidation.
zz_md_vassal_swap_enabled = { always = yes }

zz_md_exchange_structural_pair_valid = {
    OR = {
        zz_md_own_structural_pair_valid = yes
        AND = {
            zz_md_is_vassal_exchange = yes
            zz_swap_can_execute = yes
            scope:zz_md_a = scope:zz_md_prepared_a
            scope:zz_md_b = scope:zz_md_prepared_b
        }
    }
}

zz_md_is_vassal_exchange = {
    exists = scope:zz_md_vassal_actor
    scope:actor = scope:zz_md_vassal_actor
    NOT = { scope:actor = scope:recipient }
}
