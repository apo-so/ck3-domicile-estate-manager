# ===================================================
# 交换庄园 (swap domicile) - 类型/等级匹配判定 (步骤 1)
# 调用方: common/character_interactions/zz_swap_domicile_interaction.txt
# 可用 scope: scope:actor (发起者/玩家), scope:recipient (目标封臣)
# 重要: has_domicile_building_or_higher 是 domicile 作用域触发器,
#        必须用 scope:X = { domicile ?= { ... } } 形式 (已核对本体).
# ===================================================

# 同类型判定: actor 与 recipient 的庄园类型一致
zz_domicile_type_matches = {
    OR = {
        AND = {
            scope:actor    = { domicile ?= { is_domicile_type = estate } }
            scope:recipient = { domicile ?= { is_domicile_type = estate } }
        }
        AND = {
            scope:actor    = { domicile ?= { is_domicile_type = east_asian_estate } }
            scope:recipient = { domicile ?= { is_domicile_type = east_asian_estate } }
        }
        AND = {
            scope:actor    = { domicile ?= { is_domicile_type = japanese_manor } }
            scope:recipient = { domicile ?= { is_domicile_type = japanese_manor } }
        }
    }
}

zz_swap_can_execute = {
    zz_md_vassal_swap_enabled = yes
    exists = scope:actor exists = scope:recipient
    scope:actor = { is_alive = yes exists = domicile NOT = { has_variable = zz_swap_incomplete } }
    scope:recipient = { is_alive = yes this != scope:actor is_vassal_or_below_of = scope:actor exists = domicile NOT = { has_variable = zz_swap_incomplete } }
    zz_domicile_type_matches = yes zz_domicile_tier_matches = yes
    exists = scope:zz_md_prepared_a exists = scope:zz_md_prepared_b
    scope:actor = { domicile = scope:zz_md_prepared_a }
    scope:recipient = { domicile = scope:zz_md_prepared_b }
}

zz_swap_supported_types = {
    scope:actor = { domicile = { OR = { is_domicile_type = estate is_domicile_type = east_asian_estate is_domicile_type = japanese_manor } } }
    scope:recipient = { domicile = { OR = { is_domicile_type = estate is_domicile_type = east_asian_estate is_domicile_type = japanese_manor } } }
}
