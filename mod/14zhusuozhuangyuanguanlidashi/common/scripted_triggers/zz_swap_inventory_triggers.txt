# The actor and recipient use separate lists for the duration of one confirmed exchange.
zz_swap_snapshot_has = {
    is_target_in_variable_list = { name = zz_swap_building_snapshot target = flag:$BUILDING$ }
}

zz_swap_building_matches = {
    trigger_if = {
        limit = { $SOURCE$ = { zz_swap_snapshot_has = { BUILDING = $BUILDING$ } } }
        has_domicile_building = $BUILDING$
    }
    trigger_else = { NOT = { has_domicile_building = $BUILDING$ } }
}

zz_swap_needs_rebuild = {
    NOT = { has_domicile_building = $BUILDING$ }
    $SOURCE$ = { zz_swap_snapshot_has = { BUILDING = $BUILDING$ } }
}

# Recheck the existing interaction contract against the two domiciles frozen at confirmation.
zz_swap_can_commit = {
    exists = scope:actor
    exists = scope:recipient
    exists = scope:zz_swap_a
    exists = scope:zz_swap_b
    NOT = { scope:zz_swap_a = scope:zz_swap_b }
    scope:actor = { is_alive = yes domicile = scope:zz_swap_a }
    scope:recipient = {
        is_alive = yes
        this != scope:actor
        is_vassal_or_below_of = scope:actor
        domicile = scope:zz_swap_b
    }
    scope:zz_swap_a = { is_domicile_type = $TYPE$ }
    zz_domicile_type_matches = yes
    zz_domicile_tier_matches = yes
}
