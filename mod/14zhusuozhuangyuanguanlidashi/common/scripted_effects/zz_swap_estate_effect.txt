zz_swap_estate_effect = {
    scope:actor = { clear_variable_list = zz_swap_building_snapshot }
    scope:recipient = { clear_variable_list = zz_swap_building_snapshot }
    scope:zz_swap_a = { zz_swap_capture_estate = { SOURCE = scope:actor } }
    scope:zz_swap_b = { zz_swap_capture_estate = { SOURCE = scope:recipient } }

    scope:zz_swap_a = { zz_swap_apply_estate = { SOURCE = scope:recipient } }
    scope:zz_swap_b = { zz_swap_apply_estate = { SOURCE = scope:actor } }
    if = {
        limit = {
            scope:zz_swap_a = { zz_swap_matches_estate = { SOURCE = scope:recipient } }
            scope:zz_swap_b = { zz_swap_matches_estate = { SOURCE = scope:actor } }
        }
        trigger_event = zz_swap_domicile.0100
    }
    else = {
        # Use the original snapshots to restore both sides, then check the result.
        scope:zz_swap_a = { zz_swap_apply_estate = { SOURCE = scope:actor } }
        scope:zz_swap_b = { zz_swap_apply_estate = { SOURCE = scope:recipient } }
        if = {
            limit = {
                scope:zz_swap_a = { zz_swap_matches_estate = { SOURCE = scope:actor } }
                scope:zz_swap_b = { zz_swap_matches_estate = { SOURCE = scope:recipient } }
            }
            trigger_event = zz_swap_domicile.0102
        }
        else = { trigger_event = zz_swap_domicile.0103 }
    }
    scope:actor = { clear_variable_list = zz_swap_building_snapshot }
    scope:recipient = { clear_variable_list = zz_swap_building_snapshot }
}
