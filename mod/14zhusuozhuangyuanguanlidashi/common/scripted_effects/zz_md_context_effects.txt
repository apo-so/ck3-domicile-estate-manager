# Only mod UI references are changed; no title or domicile is created here.
zz_md_bind_selected_effect = {
    if = {
        limit = { exists = scope:selected_title }
        scope:selected_title = {
            remove_variable = zz_md_view_domicile
            if = { limit = { exists = holder exists = title_domicile } set_variable = { name = zz_md_view_domicile value = title_domicile } }
        }
    }
}
zz_md_refresh_effect = {
    clear_variable_list = zz_md_titles
    set_variable = { name = zz_md_title_count value = 0 }
    every_held_title = {
        remove_variable = zz_md_view_domicile
        if = {
            limit = { exists = title_domicile holder = root }
            set_variable = { name = zz_md_view_domicile value = title_domicile }
            save_scope_as = zz_md_row_title
            root = { add_to_variable_list = { name = zz_md_titles target = scope:zz_md_row_title } change_variable = { name = zz_md_title_count add = 1 } }
            set_variable = { name = zz_md_display_number value = root.var:zz_md_title_count }
        }
    }
}
zz_md_open_request_effect = {
    if = {
        limit = { zz_md_validation_allowed = yes }
        remove_variable = zz_md_last_requested_title
        remove_variable = zz_md_last_requested_domicile
        set_variable = { name = zz_md_last_request_status value = flag:invalid }
    }
    remove_variable = zz_md_open_title
    remove_variable = zz_md_open_domicile
    zz_md_bind_selected_effect = yes
    if = {
        limit = { zz_md_selected_bound = yes }
        set_variable = { name = zz_md_open_title value = scope:selected_title }
        set_variable = { name = zz_md_open_domicile value = scope:selected_title.title_domicile }
        if = {
            limit = { zz_md_validation_allowed = yes }
            set_variable = { name = zz_md_last_requested_title value = scope:selected_title }
            set_variable = { name = zz_md_last_requested_domicile value = scope:selected_title.title_domicile }
            set_variable = { name = zz_md_last_request_status value = flag:requested }
        }
    }
}
zz_md_open_manager_effect = { zz_md_refresh_effect = yes set_variable = { name = zz_md_manager_visible value = yes } }
