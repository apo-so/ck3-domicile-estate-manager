                scope:zz_swap_a = { zz_swap_matches_japanese_manor = { SOURCE = scope:actor } }
                scope:zz_swap_b = { zz_swap_matches_japanese_manor = { SOURCE = scope:recipient } }
            }
            trigger_event = zz_swap_domicile.0102
        }
        else = { trigger_event = zz_swap_domicile.0103 }
