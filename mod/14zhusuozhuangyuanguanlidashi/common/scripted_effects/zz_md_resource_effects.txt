zz_md_merge_provisions_pair_effect = {
    if = {
        limit = { zz_md_resource_pair_valid = yes scope:zz_md_res_source = { is_domicile_type = camp } }
        set_variable = { name = zz_md_res_source value = scope:zz_md_res_source }
        set_variable = { name = zz_md_res_target value = scope:zz_md_res_target }
        set_variable = { name = zz_md_res_s_before value = scope:zz_md_res_source.provisions }
        set_variable = { name = zz_md_res_t_before value = scope:zz_md_res_target.provisions }
        set_variable = { name = zz_md_res_requested value = var:zz_md_res_s_before }
        set_variable = { name = zz_md_res_received value = 0 }
        set_variable = { name = zz_md_res_moved value = 0 }
        set_variable = { name = zz_md_res_remaining value = var:zz_md_res_s_before }
        set_variable = { name = zz_md_res_kind value = flag:provisions }
        zz_md_resource_ledger_begin_effect = yes
        save_scope_value_as = { name = zz_md_res_required_total value = { value = var:zz_md_res_s_before add = var:zz_md_res_t_before } }
        if = {
            limit = { var:zz_md_res_s_before = 0 }
            zz_md_resource_notice_zero_effect = yes
        }
        else_if = {
            limit = { scope:zz_md_res_target = { max_provisions >= scope:zz_md_res_required_total } var:zz_md_res_s_before > 0 var:zz_md_res_t_before >= 0 }
            # Credit first; only an observed positive credit may authorize a source debit.
            scope:zz_md_res_target = { change_provisions = root.var:zz_md_res_requested }
            zz_md_resource_observe_credit_effect = yes
            set_variable = { name = zz_md_res_phase value = flag:after_credit }
            set_variable = { name = zz_md_res_received value = { value = scope:zz_md_res_target.provisions subtract = var:zz_md_res_t_before } }
            if = {
                limit = { zz_md_resource_pair_valid = yes var:zz_md_res_received >= 0 var:zz_md_res_received <= var:zz_md_res_requested }
                if = {
                    limit = { var:zz_md_res_received > 0 }
                    set_variable = { name = zz_md_res_phase value = flag:debit_attempt }
                    scope:zz_md_res_source = { change_provisions = { value = root.var:zz_md_res_received multiply = -1 } }
                }
                set_variable = { name = zz_md_res_moved value = { value = var:zz_md_res_s_before subtract = scope:zz_md_res_source.provisions } }
                set_variable = { name = zz_md_res_phase value = flag:after_debit }
                if = {
                    limit = { zz_md_resource_pair_valid = yes var:zz_md_res_moved >= 0 var:zz_md_res_moved <= var:zz_md_res_received }
                    # Withdraw only excess target credit if the source debit was partial/refused.
                    if = {
                        limit = { var:zz_md_res_moved < var:zz_md_res_received }
                        set_variable = { name = zz_md_res_compensation_requested value = { value = var:zz_md_res_received subtract = var:zz_md_res_moved } }
                        set_variable = { name = zz_md_res_phase value = flag:compensation_attempt }
                        scope:zz_md_res_target = { change_provisions = { value = root.var:zz_md_res_moved subtract = root.var:zz_md_res_received } }
                    }
                    set_variable = { name = zz_md_res_phase value = flag:postcheck }
                    set_variable = { name = zz_md_res_remaining value = scope:zz_md_res_source.provisions }
                    save_scope_value_as = { name = zz_md_res_expected_source value = { value = var:zz_md_res_s_before subtract = var:zz_md_res_moved } }
                    save_scope_value_as = { name = zz_md_res_expected_target value = { value = var:zz_md_res_t_before add = var:zz_md_res_moved } }
                    if = {
                        limit = {
                            zz_md_resource_pair_valid = yes
                            scope:zz_md_res_source = { provisions = scope:zz_md_res_expected_source }
                            scope:zz_md_res_target = { provisions = scope:zz_md_res_expected_target }
                        }
                        if = { limit = { var:zz_md_res_moved = var:zz_md_res_requested } zz_md_resource_notice_done_effect = yes }
                        else = { zz_md_resource_notice_partial_effect = yes }
                    }
                    else = { zz_md_resource_fail_effect = yes }
                }
                else = { zz_md_resource_fail_effect = yes }
            }
            else = { zz_md_resource_fail_effect = yes }
        }
        else = { zz_md_resource_notice_capacity_effect = yes }
        zz_md_resource_collect_pair_effect = yes
    }
}

zz_md_resource_fail_effect = {
    set_variable = { name = zz_md_res_failed value = yes }
    # Retain the first failed pair and actual/unknown readback. Never blindly retry unknown loss.
    if = {
        limit = { NOT = { has_variable = zz_md_res_failure_captured } }
        set_variable = { name = zz_md_res_failure_phase value = var:zz_md_res_phase }
        zz_md_resource_read_final_effect = yes
        set_variable = { name = zz_md_res_failure_captured value = yes }
        zz_md_resource_notice_failed_effect = yes
    }
}
zz_md_merge_resources_effect = {
    if = {
        limit = { zz_md_resource_merge_available = yes }
        set_variable = { name = zz_md_res_busy value = yes }
        set_variable = { name = zz_md_res_batch value = yes }
        set_variable = { name = zz_md_res_total_requested value = 0 }
        set_variable = { name = zz_md_res_total_moved value = 0 }
        set_variable = { name = zz_md_res_total_remaining value = 0 }
        set_variable = { name = zz_md_res_pair_count value = 0 }
        set_variable = { name = zz_md_res_totals_known value = yes }
        remove_variable = zz_md_res_batch_target_before
        if = { limit = { domicile = { is_domicile_type = camp } } set_variable = { name = zz_md_res_batch_target_before value = domicile.provisions } }
        else_if = { limit = { domicile = { is_domicile_type = yurt } } set_variable = { name = zz_md_res_batch_target_before value = domicile.herd } }
        remove_variable = zz_md_res_target_title_found
        domicile = { save_scope_as = zz_md_res_target }
        every_held_title = {
            limit = { exists = title_domicile title_domicile = root.domicile }
            save_scope_as = zz_md_notice_target_title
            root = { set_variable = { name = zz_md_res_target_title_found value = yes } }
        }
        if = {
            limit = { has_variable = zz_md_res_target_title_found }
            every_held_title = {
                limit = { holder = root exists = title_domicile title_domicile != root.domicile }
                save_scope_as = zz_md_notice_source_title
                title_domicile = { save_scope_as = zz_md_res_source }
                root = {
                    if = {
                        limit = { NOT = { has_variable = zz_md_res_failed } zz_md_resource_pair_valid = yes }
                        if = { limit = { scope:zz_md_res_source = { is_domicile_type = camp } } zz_md_merge_provisions_pair_effect = yes }
                        else_if = { limit = { scope:zz_md_res_source = { is_domicile_type = yurt } } zz_md_merge_herd_pair_effect = yes }
                    }
                }
            }
        }
        zz_md_resource_batch_notice_effect = yes
        remove_variable = zz_md_res_batch
        remove_variable = zz_md_res_notice_requested
        if = {
            limit = { NOT = { has_variable = zz_md_res_failed } }
            remove_variable = zz_md_res_busy
            remove_variable = zz_md_res_target_title_found
        }
    }
}

# Message text is generated inside this effect, using per-message saved scalars.
# It never renders a later value from the character's mutable transaction variables.
zz_md_resource_notice_effect = {
    if = {
        limit = { OR = { NOT = { has_variable = zz_md_res_batch } AND = { debug_only = yes zz_md_validation_allowed = yes } } }
    send_interface_message = {
        type = $TYPE$
        left_icon = this
        if = { limit = { var:zz_md_res_kind = flag:provisions } custom_tooltip = zz_md_provisions }
        else_if = { limit = { var:zz_md_res_kind = flag:herd } custom_tooltip = zz_md_herd }
        else = { custom_tooltip = zz_md_resource_kind_unknown_desc }
        if = {
            limit = { exists = scope:zz_md_notice_source_title exists = scope:zz_md_notice_target_title }
            scope:zz_md_notice_source_title = { save_scope_as = zz_md_message_source_title }
            scope:zz_md_notice_target_title = { save_scope_as = zz_md_message_target_title }
            custom_tooltip = zz_md_resource_pair_desc
        }
        else = { custom_tooltip = zz_md_resource_titles_unknown_desc }
        if = {
            limit = { has_variable = zz_md_res_failed }
            if = { limit = { debug_only = yes zz_md_validation_allowed = yes } zz_md_resource_failure_message_fields_effect = yes }
        }
        else = {
            save_scope_value_as = { name = zz_md_message_requested value = var:zz_md_res_requested }
            save_scope_value_as = { name = zz_md_message_received value = var:zz_md_res_received }
            save_scope_value_as = { name = zz_md_message_moved value = var:zz_md_res_moved }
            save_scope_value_as = { name = zz_md_message_remaining value = var:zz_md_res_remaining }
            custom_tooltip = zz_md_resource_numbers_desc
        }
        custom_tooltip = $STATUS$
    }
    }
}

# Character-owned, first-failure ledger. Missing final fields mean UNKNOWN, never zero.
zz_md_resource_ledger_begin_effect = {
    if = {
        limit = { NOT = { has_variable = zz_md_res_failed } }
        remove_variable = zz_md_res_failure_captured
        remove_variable = zz_md_res_failure_phase
        remove_variable = zz_md_res_t_after_credit
        remove_variable = zz_md_res_source_final
        remove_variable = zz_md_res_target_final
        remove_variable = zz_md_res_source_final_known
        remove_variable = zz_md_res_target_final_known
        remove_variable = zz_md_res_actual_debit
        remove_variable = zz_md_res_actual_credit
        remove_variable = zz_md_res_net_delta
        remove_variable = zz_md_res_compensation_returned
        set_variable = { name = zz_md_res_compensation_requested value = 0 }
        set_variable = { name = zz_md_res_phase value = flag:before_credit }
    }
}

            if = {
                limit = { has_variable = zz_md_res_totals_known exists = domicile has_variable = zz_md_res_batch_target_before }

                save_scope_value_as = { name = zz_md_message_total_moved value = var:zz_md_res_total_moved }
                save_scope_value_as = { name = zz_md_message_total_remaining value = var:zz_md_res_total_remaining }

            }
            else = { custom_tooltip = zz_md_resource_summary_unknown }
            if = { limit = { has_variable = zz_md_res_failed } custom_tooltip = zz_md_resource_failed_desc }
        }
